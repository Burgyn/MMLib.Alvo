SET LOCAL lock_timeout = '5s'
;
CREATE TABLE work_orders (
    id uuid NOT NULL,
    tenant_id uuid NOT NULL,
    title text NOT NULL,
    CONSTRAINT "PK_work_orders" PRIMARY KEY (id),
    CONSTRAINT "AK_work_orders_tenant_id_id" UNIQUE (tenant_id, id)
);

;
CREATE TABLE notes (
    id uuid NOT NULL,
    tenant_id uuid NOT NULL,
    work_order_id uuid,
    CONSTRAINT "PK_notes" PRIMARY KEY (id),
    CONSTRAINT "FK_notes_work_orders_tenant_id_work_order_id" FOREIGN KEY (tenant_id, work_order_id) REFERENCES work_orders (tenant_id, id) ON DELETE SET NULL (work_order_id)
);

;
CREATE TABLE tasks (
    id uuid NOT NULL,
    tenant_id uuid NOT NULL,
    work_order_id uuid NOT NULL,
    CONSTRAINT "PK_tasks" PRIMARY KEY (id),
    CONSTRAINT "FK_tasks_work_orders_tenant_id_work_order_id" FOREIGN KEY (tenant_id, work_order_id) REFERENCES work_orders (tenant_id, id) ON DELETE CASCADE
);

;
CREATE TABLE visits (
    id uuid NOT NULL,
    tenant_id uuid NOT NULL,
    work_order_id uuid NOT NULL,
    CONSTRAINT "PK_visits" PRIMARY KEY (id),
    CONSTRAINT "FK_visits_work_orders_tenant_id_work_order_id" FOREIGN KEY (tenant_id, work_order_id) REFERENCES work_orders (tenant_id, id) ON DELETE RESTRICT
);

;
CREATE INDEX "IX_notes_tenant_id_work_order_id" ON notes (tenant_id, work_order_id);

;
CREATE INDEX "IX_tasks_tenant_id_work_order_id" ON tasks (tenant_id, work_order_id);

;
CREATE INDEX "IX_visits_tenant_id_work_order_id" ON visits (tenant_id, work_order_id);
